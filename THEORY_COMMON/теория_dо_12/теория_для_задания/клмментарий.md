# kustomization.yaml — главный файл базовой конфигурации Kustomize.
# Перечисляет все YAML-файлы, которые входят в эту "базу" (то, что 
# одинаково для всех окружений: dev, staging, production).
#
# - apiVersion и kind — обязательные маркеры, что это файл Kustomize
# - resources — список манифестов, которые будут объединены в одну сборку
apiVersion: kustomize.config.k8s.io/v1beta1
kind: Kustomization

resources:
  - 03-postgres-pv.yaml      # PV+PVC для PostgreSQL
  - 04-postgres.yaml         # сам PostgreSQL
  - 05-rabbitmq.yaml         # очередь сообщений
  - 06-session-service.yaml  # авторизация
  - 07-hotel-service.yaml    # справочник отелей
  - 08-booking-service.yaml  # бронирования
  - 09-payment-service.yaml  # платежи
  - 10-loyalty-service.yaml  # программа лояльности
  - 11-report-service.yaml   # статистика
  - 12-gateway-service.yaml  # API Gateway (главная точка входа)
  - 13-app-ingress.yaml      # внешняя точка доступа через Nginx
